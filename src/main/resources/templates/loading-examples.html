💻 Code Examples

// Show simple content loading
const container = document.querySelector('.main-content-area');
Loading.showContentLoading(container, 'Loading data...', 'lg');

// Do async work
const data = await fetchData();

// Hide loading
Loading.hideContentLoading(container);
// Show skeleton loaders
const container = document.querySelector('.main-content-area');
Loading.showSkeleton(container, 'table');

// Load data
const assets = await fetchAssets();
renderTable(assets);

// Hide skeleton to reveal content
Loading.hideSkeleton(container);
// Advanced: Spinner + Skeleton
try {
  const container = document.querySelector('.main-content-area');
  Loading.showContentLoading(container, 'Fetching records...', 'lg');

  const data = await fetch('/api/assets').then(r => r.json());

  Loading.hideContentLoading(container);
  renderContent(data);
} catch (error) {
  Loading.hideContentLoading(container);
  showErrorMessage('Failed to load data');
}
